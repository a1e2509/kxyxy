    <!-- 引入 Three.js 核心库和扩展库 -->
    <script type="importmap">
        {
            "imports": {
                "three": "https://unpkg.com/three@0.128.0/build/three.module.js",
                "three/addons/": "https://unpkg.com/three@0.128.0/examples/jsm/"
            }
        }
    </script>

    <script type="module">
        import * as THREE from 'three';
        import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

        // --- 初始化场景、相机、渲染器 ---
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x000000); // 黑色背景，突出粉色粒子

        // 使用透视相机，适合移动端视野
        const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 2, 8);
        camera.lookAt(0, 1.5, 0);

        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); // 限制像素比，优化性能
        document.body.appendChild(renderer.domElement);

        // --- 添加一些环境光微弱效果，使粒子颜色更鲜艳 (对Points材质影响不大，但有助于整体感知) ---
        // 实际上Points材质不需要光照，但为了氛围可以加一点环境光让相机稍微感知，这里省略

        // --- 添加简单轨道控制，方便查看，但移动端可能不需要，为了体验保留，并允许用户旋转查看立体效果 ---
        const controls = new OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;
        controls.autoRotate = true;
        controls.autoRotateSpeed = 1.0;
        controls.enableZoom = true;
        controls.enablePan = false;
        controls.target.set(0, 1.5, 0);
        controls.maxPolarAngle = Math.PI / 2; // 限制不要转到地下

        // --- 生成圣诞树粒子数据 ---
        // 参数配置
        const treeHeight = 4.5;          // 树总高度
        const baseRadius = 1.8;          // 底部半径
        const topRadius = 0.1;           // 顶部半径
        const particleCount = 5500;      // 总粒子数，保证移动端性能流畅
        const pinkHues = [0.92, 0.95, 0.88, 0.9, 0.98]; // 粉色色相范围（0.9左右为粉红，0.95为浅粉）

        // 存储粒子的位置和颜色
        const positions = new Float32Array(particleCount * 3);
        const colors = new Float32Array(particleCount * 3);

        // 用于动画的“生成进度”：每个粒子有一个y轴比例阈值，从0到1
        // 但我们通过后期动画控制显示进度，这里先构建所有粒子位置和颜色
        // 为了让“从下到上依次生成”效果自然，粒子在生成时按照高度排序，这样进度控制时就会从底部先出现
       
        // 先创建数组用于排序
        const tempParticles = [];

        for (let i = 0; i < particleCount; i++) {
            // 1. 决定粒子在树上的高度 (y坐标)
            // 为了让底部密集、顶部稀疏符合圆锥体积，使用sqrt分布让更多粒子靠近底部
            // 但为了视觉均匀，也可以直接均匀分布高度并让半径随高度线性减小。这里使用均匀高度分布，但通过半径变化来表现体积。
            // 为了底部更丰富，采用随机但倾向于底部的分布：rand^0.7 会使得值偏向0（底部）
            const randY = Math.pow(Math.random(), 0.7); // 0~1，偏向底部
            const y = randY * treeHeight; // 0 到 treeHeight
           
            // 2. 根据高度计算当前层的半径 (线性插值)
            const t = y / treeHeight; // 0底部 -> 1顶部
            const radius = baseRadius * (1 - t) + topRadius * t;
           
            // 3. 随机角度，生成圆环上的点
            const angle = Math.random() * Math.PI * 2;
           
            // 4. 半径上加入随机扰动，使树看起来更自然，不是严格的圆锥面
            // 扰动幅度随高度减小，顶部更收紧
            const randomRad = radius * (0.7 + Math.random() * 0.6); // 0.7~1.3倍半径范围
            // 但为了树形状保持，稍微限制顶部扰动
            const finalRadius = Math.min(randomRad, radius * 1.2);
           
            const x = Math.cos(angle) * finalRadius;
            const z = Math.sin(angle) * finalRadius;
           
            // 轻微的高度扰动，让层次更自然
            const yOffset = (Math.random() - 0.5) * 0.2;
            const finalY = Math.max(0, Math.min(treeHeight, y + yOffset));
           
            // 5. 颜色：粉色系，根据高度和随机变化亮度与饱和度
            // 基础色相取粉色范围
            const hue = pinkHues[Math.floor(Math.random() * pinkHues.length)];
            // 饱和度：底部稍微深一点，顶部浅一点，但保持粉色感觉
            const saturation = 0.7 + Math.random() * 0.3; // 0.7~1.0
            // 亮度：让粒子有明暗变化，增加立体感
            const lightness = 0.6 + Math.random() * 0.4; // 0.6~1.0
           
            // 将HSL转为RGB (使用Three.js的Color)
            const color = new THREE.Color().setHSL(hue, saturation, lightness);
           
            // 存储到临时数组，用于按高度排序
            tempParticles.push({
                x, y: finalY, z,
                r: color.r, g: color.g, b: color.b,
                // 存储原始高度用于排序和进度
                heightRatio: finalY / treeHeight // 0~1
            });
        }

        // 按高度（y值）从小到大排序，确保生成动画从下往上
        tempParticles.sort((a, b) => a.y - b.y);

        // 填充到最终数组
        for (let i = 0; i < particleCount; i++) {
            const p = tempParticles[i];
            positions[i * 3] = p.x;
            positions[i * 3 + 1] = p.y;
            positions[i * 3 + 2] = p.z;
           
            colors[i * 3] = p.r;
            colors[i * 3 + 1] = p.g;
            colors[i * 3 + 2] = p.b;
        }

        // --- 创建粒子几何体 ---
        const geometry = new THREE.BufferGeometry();
        geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
        geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

        // --- 创建圆形粒子纹理 (用于点材质，让粒子呈现柔和圆点) ---
        function createParticleTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 32;
            canvas.height = 32;
            const ctx = canvas.getContext('2d');
           
            // 绘制径向渐变圆点
            const gradient = ctx.createRadialGradient(16, 16, 0, 16, 16, 16);
            gradient.addColorStop(0, 'rgba(255,255,255,1)');
            gradient.addColorStop(0.5, 'rgba(255,200,220,1)'); // 粉色光晕
            gradient.addColorStop(1, 'rgba(255,105,180,0)');
           
            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, 32, 32);
           
            return new THREE.CanvasTexture(canvas);
        }

        const particleTexture = createParticleTexture();

        // --- 点材质 ---
        const material = new THREE.PointsMaterial({
            size: 0.12,
            map: particleTexture,
            vertexColors: true,      // 使用每个粒子的颜色
            transparent: true,
            blending: THREE.AdditiveBlending, // 叠加混合，产生发光效果
            depthWrite: false,
            sizeAttenuation: true
        });

        // 创建Points对象
        const treeParticles = new THREE.Points(geometry, material);
        scene.add(treeParticles);

        // --- 添加一些漂浮的背景星星，增强氛围 (可选，不影响主体) ---
        const starGeometry = new THREE.BufferGeometry();
        const starCount = 600;
        const starPositions = new Float32Array(starCount * 3);
        for (let i = 0; i < starCount; i++) {
            // 随机分布在较大范围内
            starPositions[i*3] = (Math.random() - 0.5) * 30;
            starPositions[i*3+1] = (Math.random() - 0.5) * 20 + 2;
            starPositions[i*3+2] = (Math.random() - 0.5) * 30 - 5;
        }
        starGeometry.setAttribute('position', new THREE.BufferAttribute(starPositions, 3));
        const starMaterial = new THREE.PointsMaterial({
            color: 0xffaacc,
            size: 0.08,
            transparent: true,
            blending: THREE.AdditiveBlending,
            depthWrite: false,
            sizeAttenuation: true
        });
        const stars = new THREE.Points(starGeometry, starMaterial);
        scene.add(stars);

        // --- 动画控制：从下到上依次生成 ---
        // 思路：每个粒子有一个基于高度的“出现阈值”，我们通过修改PointsMaterial的opacity或者修改geometry的drawRange？
        // 但drawRange无法实现按高度过滤。更好的办法：通过顶点着色器控制每个粒子的透明度？但为了简单且兼容移动端，
        // 我们可以使用一个简单的方法：在动画循环中，逐步增加一个进度值，然后重新构建几何体？那太消耗性能。
        // 替代方案：使用多个Points？或者使用一种技巧：为每个粒子设置一个自定义属性表示其出现时机，然后在着色器中丢弃？但需要修改材质。
        // 为了保持简单且性能好，这里采用一个“视觉欺骗”的方法：粒子实际上全部存在，但我们将点大小设置为0（不可见），
        // 然后随着进度增加，按高度顺序逐渐将点大小恢复？但每个粒子大小是统一的，不能单独控制。
        // 因此，我们换一种方式：使用“缩放”整个树？但无法实现从下到上。
       
        // 最终方案：使用着色器或者干脆在动画循环中根据高度逐步显示粒子，具体实现：
        // 我们重新创建两个几何体？太复杂。我们采用一个技巧：因为粒子已按高度排序，我们可以通过更新geometry的drawRange来逐步显示更多的粒子索引。
        // 由于顶点缓冲区中的顺序就是按高度排序的，所以drawRange(0, count) 会显示前count个粒子，这些粒子恰好是高度较低的！
        // 完美！这样我们只需要在动画中递增count即可实现从下往上出现。
       
        // 初始时只渲染0个粒子（drawRange从0开始）
        geometry.setDrawRange(0, 0);
       
        // 动画参数
        let progress = 0; // 0 到 1
        const animationSpeed = 0.005; // 每帧增加的速度，约3秒完成
       
        // 存储总粒子数
        const totalCount = particleCount;
       
        // 为了让效果更柔和，还控制粒子大小从0逐渐变大？但drawRange已经够用，不过可以添加一个整体大小变化。
        // 也可以让粒子大小稍微随进度变化，增加生动感。
       
        // 在动画循环中
        function animate() {
            requestAnimationFrame(animate);
           
            // 更新进度，直到1
            if (progress < 1) {
                progress += animationSpeed;
                if (progress > 1) progress = 1;
               
                // 根据进度计算当前应显示的粒子数量
                // 使用缓动函数让开始时慢，中间快，结束时慢？也可以简单线性
                // 这里使用easeOutQuad让生成过程更自然
                const easedProgress = 1 - Math.pow(1 - progress, 1.5); // 稍微带点缓动
                const currentCount = Math.floor(easedProgress * totalCount);
               
                // 更新drawRange，只显示前currentCount个粒子（即高度从低到高的粒子）
                geometry.setDrawRange(0, currentCount);
               
                // 同时微调粒子大小，让整体看起来更有生长感（可选）
                // 但为了性能，不每帧修改material.size，或者只在一定范围内变化
                // material.size = 0.08 + 0.08 * easedProgress; // 从0.08到0.16
            }
           
            // 自动旋转通过OrbitControls的autoRotate实现
            controls.update();
           
            // 让背景星星缓慢旋转
            stars.rotation.y += 0.0002;
            stars.rotation.x += 0.0001;
           
            // 粒子树的整体稍微旋转也可以，但OrbitControls自动旋转已够
           
            renderer.render(scene, camera);
        }
       
        animate();

        // --- 窗口大小变化适配 ---
        window.addEventListener('resize', onWindowResize, false);
        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // 启动后稍微延迟一下，让初始效果更明显（可选，但不必）
        // 为移动端微信，确保全屏
        console.log('3D粉色圣诞树已启动！');
    </script>
